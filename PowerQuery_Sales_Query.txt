let
    Source = Excel.Workbook(File.Contents("Online Retail.xlsx"), null, true),
    Sheet = Source{[Item="Sheet1",Kind="Sheet"]}[Data],
    PromotedHeaders = Table.PromoteHeaders(Sheet, [PromoteAllScalars=true]),
    ChangedType = Table.TransformColumnTypes(PromotedHeaders,{{"InvoiceDate", type datetime},{"Quantity", Int64.Type},{"UnitPrice", type number},{"CustomerID", type text}}),
    RemovedBlank = Table.SelectRows(ChangedType, each [CustomerID] <> null and [Description] <> null),
    RemovedInvalid = Table.SelectRows(RemovedBlank, each [Quantity] > 0 and [UnitPrice] > 0),
    AddedSales = Table.AddColumn(RemovedInvalid, "Sales Amount", each [Quantity] * [UnitPrice], type number),
    RemovedDuplicates = Table.Distinct(AddedSales)
in
    RemovedDuplicates
